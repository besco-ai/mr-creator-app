def criar_site_empresa(cnpj: str) -> dict:
    """Cria site empresarial via API (consome 1 token).
    NOVO: Valida que os dados retornados incluem endereço COMPLETO."""
    res = _api_post('criar_site', {'cnpj': cnpj}, timeout=120)
    
    # NOVO: Validar que o endereço retornado é COMPLETO
    if res.get('sucesso'):
        empresa = res.get('empresa', {})
        tem_logradouro = bool(empresa.get('logradouro', '').strip())
        tem_bairro_cidade_cep = (
            bool(empresa.get('bairro', '').strip()) and
            bool(empresa.get('municipio', '').strip()) and
            bool(empresa.get('cep', '').strip())
        )
        
        if not (tem_logradouro or tem_bairro_cidade_cep):
            # Endereço incompleto — sinalizar para o bot
            res['endereco_incompleto'] = True
            logger.warning('[Acesso] criar_site retornou endereço incompleto (CNPJ %s)', cnpj)
    
    return res
